using Microsoft.AspNetCore.Identity;
using Microsoft.Extensions.DependencyInjection;


public static class DbInitializer
{
    public static void Initialize(IServiceProvider serviceProvider)
    {
        using var scope = serviceProvider.CreateScope();
        var roleManager = scope.ServiceProvider.GetRequiredService<RoleManager<IdentityRole>>();
        var userManager = scope.ServiceProvider.GetRequiredService<UserManager<AppUser>>();

        string[] roles = new[] { "Admin", "Sales" };
        foreach (var role in roles)
        {
            if (!roleManager.RoleExistsAsync(role).Result)
            {
                roleManager.CreateAsync(new IdentityRole(role)).Wait();
            }
        }

        if (userManager.FindByNameAsync("admin").Result == null)
        {
            var adminUser = new AppUser
            {
                UserName = "admin",
                Email = "admin@hayaa.com",
                EmailConfirmed = true
            };
            userManager.CreateAsync(adminUser, "Admin123!").Wait();
            userManager.AddToRoleAsync(adminUser, "Admin").Wait();
        }

        if (userManager.FindByNameAsync("sales").Result == null)
        {
            var salesUser = new AppUser
            {
                UserName = "sales",
                Email = "sales@hayaa.com",
                EmailConfirmed = true
            };
            userManager.CreateAsync(salesUser, "Sales123!").Wait();
            userManager.AddToRoleAsync(salesUser, "Sales").Wait();
        }
    }
}
